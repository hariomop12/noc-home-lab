---- the rule that is doing it ----
$ nft list table ip acl   (on r1)
table ip acl {
	chain forward {
		type filter hook forward priority filter; policy accept;
		ip saddr 10.10.10.0/24 ip daddr 10.100.0.12 tcp dport 80 drop
	}
}
